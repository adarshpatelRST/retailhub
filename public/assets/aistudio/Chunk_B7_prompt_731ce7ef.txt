## 3. DETAILED IMPLEMENTATION SPECIFICATIONS

### A. WebSockets Gateway (`src/core/gateway/events.gateway.ts`)
```typescript
import {
  WebSocketGateway,
  WebSocketServer,
  OnGatewayConnection,
  OnGatewayDisconnect,
  SubscribeMessage,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Logger } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';

@WebSocketGateway({
  cors: { origin: '*' },
  pingInterval: 25000,
  pingTimeout: 20000,
})
export class EventsGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer() server: Server;
  private readonly logger = new Logger(EventsGateway.name);

  constructor(private readonly jwtService: JwtService) {}

  async handleConnection(client: Socket) {
    try {
      const token = client.handshake.auth?.token || client.handshake.headers?.authorization?.split(' ')[1];
      if (!token) {
        client.disconnect();
        return;
      }
      const payload = this.jwtService.verify(token);
      client.data.user = payload;

      // Join scoped rooms based on user assignments
      if (['EXECUTIVE', 'ADMIN'].includes(payload.role)) {
        client.join('enterprise');
      }
      if (payload.assignedRegionId) {
        client.join(`region_${payload.assignedRegionId}`);
      }
      if (payload.assignedStoreId) {
        client.join(`store_${payload.assignedStoreId}`);
      }

      this.logger.log(`Client connected: ${client.id} (User: ${payload.userId}, Role: ${payload.role})`);
    } catch (err) {
      this.logger.warn(`Unauthorized WebSocket connection attempt: ${err.message}`);
      client.disconnect();
    }
  }

  handleDisconnect(client: Socket) {
    this.logger.log(`Client disconnected: ${client.id}`);
  }

  @SubscribeMessage('client_reconnected')
  handleClientReconnected(client: Socket) {
    client.emit('sync_state_requested');
  }

  emitStockUpdated(storeId: string, productId: string, availableQuantity: number) {
    this.server.to(`store_${storeId}`).to('enterprise').emit('STOCK_UPDATED', {
      storeId,
      productId,
      availableQuantity,
      timestamp: new Date().toISOString(),
    });
  }

  emitRecCreated(recommendation: any) {
    const regionRoom = recommendation.store?.regionId ? `region_${recommendation.store.regionId}` : 'enterprise';
    this.server.to(regionRoom).to('enterprise').emit('REC_CREATED', {
      recommendation,
      timestamp: new Date().toISOString(),
    });
  }

  emitRecUpdated(recommendation: any) {
    this.server.emit('REC_UPDATED', {
      recommendationId: recommendation.id,
      status: recommendation.status,
      timestamp: new Date().toISOString(),
    });
  }
}

B. Executive Dashboard Service (src/modules/dashboard/dashboard.service.ts)
Implement consolidated SQL aggregations and caching:

getExecutiveDashboard(query: DashboardQueryDto):

Cache check: Check cache for key exec_dash_${startDate}_${endDate}. If present, return cached DTO.

Compute date range: startDate (default now - 30 days) to endDate (default now). Validate startDate <= endDate.

KPIs:

totalSales: SUM(totalAmount) from SalesTransaction.

salesTarget: Pre-configured target (e.g. ₹50,000,000).

salesAchievementPercent: (totalSales / salesTarget) * 100.

totalInventoryValue: Join Inventory and Product, calculate SUM(availableQuantity * costPrice).

stockOutSKUsCount: Count of inventory rows where availableQuantity <= reorderLevel or DOI <= 3.0.

trappedExcessCapital: SUM((availableQuantity - maximumStock) * costPrice) where availableQuantity > maximumStock.

openRecommendationsCount: Count from Recommendation where status = 'OPEN'.

Recharts Datasets:

salesTrend: Group by CAST(transactionDate AS DATE) returning daily actual revenue vs daily target.

inventoryHealthDistribution: Calculate count of items in CRITICAL, LOW, HEALTHY, and EXCESS.

topStockOutProneStores: Aggregate stock-out incidents grouped by storeId, joined with store.storeCode and store.name (limit 5).

categoryPerformance: Aggregate sales grouped by Product.categoryId, joined with Category.name.

stockOutTrend: Group prevented vs active stock-outs by week.

Exception Callouts:

Array of dynamic strings (e.g., "Central MP Region has 14 cooking oil SKUs at risk of weekend stock-out.").

Cache result for 60,000ms.

Return ExecutiveDashboardDto.

getStoreDashboard(storeId: string):

Returns store sales today vs target, Month-to-Date achievement %, local health donut counts, top 5 immediate risk items (DOI <= 2.0d), and incoming StockTransfer shipments in APPROVED_IN_TRANSIT.
