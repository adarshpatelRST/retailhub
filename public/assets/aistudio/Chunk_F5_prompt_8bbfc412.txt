### A. Generic Export Engine (`src/lib/export-utils.ts` & `ReportExportBar.tsx`)
```typescript
import jsPDF from "jspdf";
import autoTable from "jspdf-autotable";
import * as XLSX from "xlsx";

export interface ExportColumnDef<T> {
  header: string;
  accessor: keyof T | ((row: T) => string | number);
}

export function exportToPdf<T>(
  title: string,
  subtitle: string,
  columns: ExportColumnDef<T>[],
  data: T[],
  filename: string
) {
  const doc = new jsPDF({ orientation: columns.length > 7 ? "landscape" : "portrait" });
  const headers = columns.map((c) => c.header);
  const rows = data.map((row) =>
    columns.map((c) => {
      const val = typeof c.accessor === "function" ? c.accessor(row) : row[c.accessor];
      return val !== null && val !== undefined ? String(val) : "-";
    })
  );

  doc.setFontSize(16);
  doc.text(`RetailHub — ${title}`, 14, 15);
  doc.setFontSize(10);
  doc.setTextColor(100);
  doc.text(subtitle, 14, 22);

  autoTable(doc, {
    startY: 28,
    head: [headers],
    body: rows,
    styles: { fontSize: 8, cellPadding: 2 },
    headStyles: { fillColor: [37, 99, 235], textColor: 255, fontStyle: "bold" },
    alternateRowStyles: { fillColor: [248, 250, 252] },
    didDrawPage: (dataArg) => {
      const pageCount = doc.getNumberOfPages();
      doc.setFontSize(8);
      doc.text(
        `Generated on ${new Date().toLocaleString()} • Page ${dataArg.pageNumber} of ${pageCount}`,
        14,
        doc.internal.pageSize.height - 10
      );
    },
  });

  doc.save(`${filename}_${Date.now()}.pdf`);
}

export function exportToExcelOrCsv<T>(
  columns: ExportColumnDef<T>[],
  data: T[],
  filename: string,
  format: "xlsx" | "csv"
) {
  const formattedRows = data.map((row) => {
    const obj: Record<string, any> = {};
    columns.forEach((c) => {
      obj[c.header] = typeof c.accessor === "function" ? c.accessor(row) : row[c.accessor];
    });
    return obj;
  });

  const worksheet = XLSX.utils.json_to_sheet(formattedRows);
  const workbook = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(workbook, worksheet, "Report");

  if (format === "csv") {
    XLSX.writeFile(workbook, `${filename}_${Date.now()}.csv`, { bookType: "csv" });
  } else {
    XLSX.writeFile(workbook, `${filename}_${Date.now()}.xlsx`, { bookType: "xlsx" });
  }
}
